\begin{glyph}{Attach (付)}{attach}\label{glyph:attach}
Attach, in both physical and figurative senses.\\\hline
It is worth learning this 漢字 well, as it conveys the meanings of many English
words---from ``setting'' something on fire, to ``turning'' on a light or an electrical device.
\end{glyph}

\begin{comps}
\comprtwocone & 亻 + 寸 \\\hline
\comprthreecone & Person + Tiny/Inch \\\hline
\end{comps}

\begin{remglyph}
\hooglig{Attach} the \remcom{human} to the \remcom{tiny} parasite.\\\hline
\end{remglyph}

\begin{prons}
\pronsrtwocone & \textbf{フ (FU)} & \textbf{つ (tsu)}\\\hline
\rye{3}{In general the reading in the second position will be つ, and it always is
when this 漢字 is preceded by another 漢字 that is read with its verb
stem: 見付ける (み　つ{\middel}ける: ``to find'') is one example, as み is the stem
of the verb 見る.\\\hline
It is only when the preceding reading does \emph{not} form a verb with its
漢字 that 付 will sometimes becomes voiced as づ: 気付く (キ　づ{\middel}く: ``to
notice'') and 名付ける (な　づ{\middel}ける: ``to name'') are examples of this, as 気
and 名 do not form standalone verbs with キ and な in the way 見 does with
み.}
\pronsrthreecone & --- & --- \\\hline
\end{prons}

\begin{remprons}
Getting \hooglig{attached} to \comon{foosball} can stir up a \comkun{tsunami}.\\\hline
\end{remprons}
